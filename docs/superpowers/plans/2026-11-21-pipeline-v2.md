# Pipeline v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make every experiment train once, predict on many evaluation sets, and keep its per-example probabilities, so later analyses never need a retrain.

**Architecture:** The runner is split into `fit_and_predict()` (train once, predict on every named evaluation set, return one `PredictionRecord` per set) and `score()` (a scorecard row from a record, in process or loaded from disk). A corpus registry and a model registry replace the lookups hard-coded in the CLI. A `pipeline` module owns the loop and writes the prediction cache; the CLI becomes argument parsing plus printing. `run_experiment()` stays as a thin wrapper, so existing callers and tests are untouched.

**Tech Stack:** Python 3.9+, numpy, scikit-learn, pytest. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-09-25-thesis-expansion-design.md` (§2.5). Roadmap: `docs/superpowers/plans/2026-11-21-thesis-expansion-epic.md` (Plan 1, decisions D1 to D5).

```python
"""Prediction cache: the per-example probabilities of one model on one evaluation set.

An experiment used to keep only its scorecard row and throw the probabilities
away, so any new analysis (a tuned threshold, a calibration fit, a paired
significance test) needed a retrain. A :class:`PredictionRecord` keeps them:
``y_true`` and ``y_prob`` for one (model, seed, evaluation set), plus a manifest
(``meta``) saying where they came from and what the run cost. Each record is one
compressed ``.npz`` file, so analyses can be re-run from disk on any machine.

numpy + stdlib only (no torch), so this imports and tests anywhere.
"""

from __future__ import annotations

import json
import os
import re
from dataclasses import dataclass, field
from pathlib import Path

import numpy as np

# Any run of characters outside [A-Za-z0-9._] becomes a single "-". The dash is in
# the run on purpose, so "a->b" gives "a-b" rather than "a--b".
_UNSAFE = re.compile(r"[^A-Za-z0-9._]+")


@dataclass
class PredictionRecord:
    """Probabilities and ground truth for one (model, seed, evaluation set)."""

    y_true: np.ndarray   # (n_samples, n_labels) multi-hot {0, 1}
    y_prob: np.ndarray   # (n_samples, n_labels) probabilities, dtype as the model produced
    label_names: list
    meta: dict = field(default_factory=dict)

    def __post_init__(self) -> None:
        self.y_true = np.asarray(self.y_true)
        self.y_prob = np.asarray(self.y_prob)
        self.label_names = [str(name) for name in self.label_names]
        if self.y_true.shape != self.y_prob.shape:
            raise ValueError(
                f"y_true shape {self.y_true.shape} != y_prob shape {self.y_prob.shape}"
            )
        if self.y_prob.ndim != 2 or self.y_prob.shape[1] != len(self.label_names):
            raise ValueError(
                f"expected (n_samples, {len(self.label_names)}) arrays, "
                f"got shape {self.y_prob.shape}"
            )

    def __len__(self) -> int:
        return int(self.y_prob.shape[0])

    @property
    def n_labels(self) -> int:
        return len(self.label_names)


def slug(text: str) -> str:
    """Make ``text`` safe as a file-name part (``go_emotions:test`` -> ``go_emotions-test``)."""
    return _UNSAFE.sub("-", str(text)).strip("-")


def prediction_path(root, run_name: str, model: str, seed: int, eval_set: str) -> Path:
    """Where the record for (run, model, seed, evaluation set) lives under ``root``."""
    name = f"{slug(model)}__seed{int(seed)}__{slug(eval_set)}.npz"
    return Path(root) / slug(run_name) / name


def save_prediction(record: PredictionRecord, path) -> Path:
    """Write ``record`` to ``path`` (a ``.npz`` file) and return the path.

    The file is written under a temporary name and renamed into place, so an
    interrupted batch never leaves a half-written record behind. ``y_prob`` keeps
    the dtype the model produced, which makes a cached score identical to the
    in-process one.
    """
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(path.name + ".tmp")
    with open(tmp, "wb") as fh:
        np.savez_compressed(
            fh,
            y_true=record.y_true.astype(np.uint8),
            y_prob=record.y_prob,
            label_names=np.asarray(record.label_names, dtype=str),
            meta=np.asarray(json.dumps(record.meta)),
        )
    os.replace(tmp, path)
    return path


def read_meta(path) -> dict:
    """Read only the manifest of a cached record (the arrays stay on disk)."""
    with np.load(Path(path), allow_pickle=False) as archive:
        return json.loads(archive["meta"].item())


def load_prediction(path) -> PredictionRecord:
    """Read back a record written by :func:`save_prediction`."""
    with np.load(Path(path), allow_pickle=False) as archive:
        return PredictionRecord(
            y_true=archive["y_true"].astype(np.float32),
            y_prob=archive["y_prob"],
            label_names=[str(name) for name in archive["label_names"]],
            meta=json.loads(archive["meta"].item()),
        )


def list_predictions(root) -> list:
    """Manifest of every cached record under ``root``, each with its ``path``."""
    out = []
    for path in sorted(Path(root).rglob("*.npz")):
        meta = read_meta(path)
        meta["path"] = str(path)
        out.append(meta)
    return out
```

```python
"""A small noisy multi-label corpus for tests (seeded; no network, no torch)."""

import numpy as np

from emotion_classification.preprocessing import PreparedSplit

LABELS = ["joy", "anger", "sadness", "fear"]
VOCAB = {
    "joy": ["happy", "glad", "delighted", "smile"],
    "anger": ["furious", "mad", "rage", "annoyed"],
    "sadness": ["sad", "cry", "grief", "lonely"],
    "fear": ["scared", "afraid", "panic", "worried"],
}
FILLER = ["the", "today", "really", "and", "then", "it", "was", "so"]


def synthetic_split(n: int, seed: int) -> PreparedSplit:
    """``n`` texts mixing cue words for their labels with filler and some noise.

    Every text has at least one label, so no row is dropped as empty when the
    split is turned into a dataset and prepared again.
    """
    rng = np.random.RandomState(seed)
    texts, rows = [], []
    for _ in range(n):
        active = [j for j in range(len(LABELS)) if rng.rand() < 0.35]
        if not active:
            active = [int(rng.randint(len(LABELS)))]
        words = [VOCAB[LABELS[j]][rng.randint(4)] for j in active for _ in range(2)]
        words += [FILLER[rng.randint(len(FILLER))] for _ in range(4)]
        if rng.rand() < 0.2:   # noise: a cue word for a label that may not be active
            words.append(VOCAB[LABELS[rng.randint(len(LABELS))]][rng.randint(4)])
        rng.shuffle(words)
        texts.append(" ".join(words))
        row = np.zeros(len(LABELS), dtype=np.float32)
        row[active] = 1.0
        rows.append(row)
    return PreparedSplit(texts, np.vstack(rows), list(LABELS))
```

```python
"""Tests for the split runner: fit once, predict on many, score from the record.

Tiny synthetic data and scikit-learn only (no network, no torch), so it runs in CI.
"""

import time

import numpy as np
import pytest

from emotion_classification.experiment import (
    _detect_device, fit_and_predict, run_experiment, score, set_seed,
)
from emotion_classification.metrics import (
    bootstrap_f1_ci, bootstrap_per_label_f1_ci, evaluate, per_label_metrics,
)
from emotion_classification.predictions import load_prediction, save_prediction
from emotion_classification.preprocessing import PreparedSplit
from emotion_classification.scorecard import ScorecardRow
from tests.synthetic import LABELS, synthetic_split

pytest.importorskip("sklearn")

from emotion_classification.models.classical import (  # noqa: E402
    LogisticReg, NaiveBayes, RandomForest,
)

TIMING_FIELDS = {"train_seconds", "predict_latency_ms", "throughput"}


def _legacy_run_experiment(model, train, test, *, dataset_name, threshold=0.5, seed=42,
                           cost_usd=None, bootstrap=0, extra_meta=None):
    """Verbatim copy of ``run_experiment`` from before the runner was split.

    Kept as the reference the new code path must match exactly (timings aside).
    """
    set_seed(seed)

    t0 = time.perf_counter()
    model.fit(train.texts, train.Y)
    train_seconds = time.perf_counter() - t0

    t1 = time.perf_counter()
    y_prob = model.predict_proba(test.texts)
    predict_seconds = time.perf_counter() - t1
    latency_ms = (predict_seconds / max(len(test), 1)) * 1000.0
    throughput = (len(test) / predict_seconds) if predict_seconds > 0 else 0.0

    scores = evaluate(test.Y, y_prob, threshold=threshold)
    per_label = per_label_metrics(test.Y, y_prob, test.label_names, threshold)
    per_label_f1 = {name: m["f1"] for name, m in per_label.items()}
    per_label_support = {name: m["support"] for name, m in per_label.items()}

    macro_f1_ci = micro_f1_ci = None
    per_label_f1_ci = {}
    if bootstrap:
        agg = bootstrap_f1_ci(test.Y, y_prob, threshold, n_boot=bootstrap, seed=seed)
        macro_f1_ci = list(agg["macro"])
        micro_f1_ci = list(agg["micro"])
        per_label_f1_ci = {
            name: list(bounds)
            for name, bounds in bootstrap_per_label_f1_ci(
                test.Y, y_prob, test.label_names, threshold, n_boot=bootstrap, seed=seed
            ).items()
        }

    meta = {
        "n_train": len(train),
        "n_test": len(test),
        "n_labels": test.n_labels,
        "labels": test.label_names,
        "threshold": threshold,
        "seed": seed,
    }
    if extra_meta:
        meta.update(extra_meta)

    return ScorecardRow(
        model=getattr(model, "name", model.__class__.__name__),
        dataset=dataset_name,
        macro_f1=scores["macro_f1"],
        micro_f1=scores["micro_f1"],
        subset_accuracy=scores["subset_accuracy"],
        ece=scores["ece"],
        train_seconds=train_seconds,
        predict_latency_ms=latency_ms,
        throughput=throughput,
        model_size_mb=model.size_mb(),
        cost_usd=cost_usd,
        device=_detect_device(model),
        per_label_f1=per_label_f1,
        per_label_support=per_label_support,
        macro_f1_ci=macro_f1_ci,
        micro_f1_ci=micro_f1_ci,
        per_label_f1_ci=per_label_f1_ci,
        meta=meta,
    )


def _without_timings(row):
    return {k: v for k, v in row.as_dict().items() if k not in TIMING_FIELDS}


@pytest.mark.parametrize("make_model", [
    lambda: NaiveBayes(),
    lambda: LogisticReg(),
    lambda: RandomForest(n_estimators=10),

def test_run_experiment_matches_the_legacy_runner_exactly(make_model):
```

```python
"""Train a model once, predict on many evaluation sets, and score the results.

This is the glue that makes the scorecard fair: it measures *the same things the
same way* for every tier. The work is split in two so that nothing has to be
retrained to ask a new question of a run:

* :func:`fit_and_predict` seeds all RNGs, times training, then predicts on every
  named evaluation set. It returns one :class:`~emotion_classification.
  predictions.PredictionRecord` per set: probabilities, ground truth, and a
  manifest holding the timings, the model footprint and the device.
* :func:`score` turns one record into a :class:`~emotion_classification.
  scorecard.ScorecardRow` (predictive and calibration metrics, the per-emotion
  breakdown, optional bootstrap intervals). It gives the same row for a record
  just produced and for one loaded back from the prediction cache.

:func:`run_experiment` is the original one-call form (train, evaluate on one set,
return the row), kept as a thin wrapper over the two.
"""

from __future__ import annotations

import importlib
import os
import random
import time
from collections.abc import Mapping
from typing import Optional

import numpy as np

from .metrics import (
    bootstrap_f1_ci, bootstrap_per_label_f1_ci, evaluate, per_label_metrics,
)
from .models.base import EmotionModel
from .predictions import PredictionRecord
from .preprocessing import PreparedSplit
from .scorecard import ScorecardRow

DEFAULT_SEED = 42


def _detect_device(model) -> str:
    """Report the hardware a model ran on ('cpu' or e.g. 'cuda:0').

    Classical (scikit-learn) models have no torch module and run on CPU. Torch
    models expose their fitted network as ``module_`` (deep tier) or ``model_``
    (transformer tier); we read the device of its parameters. Recorded per run so
    the training-time / latency fairness caveat is visible in the results.
    """
    if importlib.util.find_spec("torch") is None:
        return "cpu"
    import torch

    for attr in ("module_", "model_"):
        module = getattr(model, attr, None)
        if isinstance(module, torch.nn.Module):
            try:
                return str(next(module.parameters()).device)
            except StopIteration:
                pass
    return "cpu"


def set_seed(seed: int = DEFAULT_SEED) -> None:
    """Seed Python, NumPy, and (if importable) PyTorch for reproducibility."""
    os.environ["PYTHONHASHSEED"] = str(seed)
    random.seed(seed)
    np.random.seed(seed)
    # Torch is optional — seed it only if the deep/transformer tiers are present.
    if importlib.util.find_spec("torch") is not None:
        import torch

        torch.manual_seed(seed)
        if torch.cuda.is_available():
            torch.cuda.manual_seed_all(seed)


def fit_and_predict(
    model: EmotionModel,
    train: PreparedSplit,
    eval_sets: "Mapping[str, PreparedSplit]",
    *,
    seed: int = DEFAULT_SEED,
    meta: Optional[dict] = None,
) -> "dict[str, PredictionRecord]":
    """Train ``model`` once on ``train``, then predict on every evaluation set.

    ``eval_sets`` maps a name (by convention ``"<corpus>:<split>"``) to a prepared
    split over the same labels as ``train``. Returns one record per set, in the
    order given. Every record carries the caller's ``meta`` plus what was measured:
    ``model``, ``seed``, ``n_train``, ``train_seconds``, ``model_size_mb``,
    ``device``, and per set ``eval_set``, ``n_eval``, ``predict_seconds``. Measured
    values win over caller keys of the same name.
    """
    if not eval_sets:
        raise ValueError("eval_sets must name at least one evaluation set")
    for name, split in eval_sets.items():
        if list(split.label_names) != list(train.label_names):
            raise ValueError(
                f"evaluation set {name!r} has labels {list(split.label_names)} but "
                f"the model is trained on {list(train.label_names)}"
            )

    set_seed(seed)

    t0 = time.perf_counter()
    model.fit(train.texts, train.Y)
    train_seconds = time.perf_counter() - t0

    predicted: dict = {}
    for name, split in eval_sets.items():
        t1 = time.perf_counter()
        y_prob = model.predict_proba(split.texts)
        predicted[name] = (y_prob, time.perf_counter() - t1)

    run_meta = dict(meta or {})
    run_meta.update({
        "model": getattr(model, "name", model.__class__.__name__),
        "seed": int(seed),
        "n_train": len(train),
        "train_seconds": float(train_seconds),
        "model_size_mb": float(model.size_mb()),
        "device": _detect_device(model),
    })

    records: dict = {}
    for name, split in eval_sets.items():
        y_prob, predict_seconds = predicted[name]
        records[name] = PredictionRecord(
            y_true=split.Y,
            y_prob=y_prob,
            label_names=list(split.label_names),
            meta={**run_meta, "eval_set": name, "n_eval": len(split),
                  "predict_seconds": float(predict_seconds)},
        )
    return records


def score(
    record: PredictionRecord,
    *,
    dataset_name: Optional[str] = None,
    threshold: float = 0.5,
    bootstrap: int = 0,
    cost_usd: Optional[float] = None,
    extra_meta: Optional[dict] = None,
) -> ScorecardRow:
    """Turn one prediction record into a scorecard row.

    Timings, model size and device come from the record's manifest, so a record
    loaded from the cache scores exactly as it did in process. ``dataset_name``
    defaults to the record's evaluation-set name. ``bootstrap`` (default 0 = off)
    is the number of resamples for confidence intervals on macro/micro-F1 and
    per-label F1.
    """
    m = record.meta
    n = len(record)
    seed = int(m.get("seed", DEFAULT_SEED))
    predict_seconds = float(m.get("predict_seconds", 0.0))
    latency_ms = (predict_seconds / max(n, 1)) * 1000.0
    throughput = (n / predict_seconds) if predict_seconds > 0 else 0.0

    scores = evaluate(record.y_true, record.y_prob, threshold=threshold)
    per_label = per_label_metrics(record.y_true, record.y_prob, record.label_names, threshold)
    per_label_f1 = {name: v["f1"] for name, v in per_label.items()}
    per_label_support = {name: v["support"] for name, v in per_label.items()}

    macro_f1_ci = micro_f1_ci = None
    per_label_f1_ci: dict = {}
    if bootstrap:
        agg = bootstrap_f1_ci(record.y_true, record.y_prob, threshold,
                              n_boot=bootstrap, seed=seed)
        macro_f1_ci = list(agg["macro"])
        micro_f1_ci = list(agg["micro"])
        per_label_f1_ci = {
            name: list(bounds)
            for name, bounds in bootstrap_per_label_f1_ci(
                record.y_true, record.y_prob, record.label_names, threshold,
                n_boot=bootstrap, seed=seed,
            ).items()
        }

    meta = {
        "n_train": m.get("n_train"),
        "n_test": n,
        "n_labels": record.n_labels,
        "labels": list(record.label_names),
        "threshold": threshold,
        "seed": seed,
    }
    if extra_meta:
        meta.update(extra_meta)

    return ScorecardRow(
        model=m.get("model", "unknown"),
        dataset=dataset_name if dataset_name is not None else m.get("eval_set", "unknown"),
        macro_f1=scores["macro_f1"],
        micro_f1=scores["micro_f1"],
        subset_accuracy=scores["subset_accuracy"],
        ece=scores["ece"],
        train_seconds=float(m.get("train_seconds", 0.0)),
        predict_latency_ms=latency_ms,
        throughput=throughput,
        model_size_mb=float(m.get("model_size_mb", 0.0)),
        cost_usd=cost_usd if cost_usd is not None else m.get("cost_usd"),
        device=m.get("device"),
        per_label_f1=per_label_f1,
        per_label_support=per_label_support,
        macro_f1_ci=macro_f1_ci,
        micro_f1_ci=micro_f1_ci,
        per_label_f1_ci=per_label_f1_ci,
        meta=meta,
    )


def run_experiment(
    model: EmotionModel,
    train: PreparedSplit,
    test: PreparedSplit,
    *,
    dataset_name: str,
    threshold: float = 0.5,
    seed: int = DEFAULT_SEED,
    cost_usd: Optional[float] = None,
    bootstrap: int = 0,
    extra_meta: Optional[dict] = None,
) -> ScorecardRow:
    """Train ``model`` on ``train``, evaluate on ``test``, return a scorecard row.

    The one-call form: :func:`fit_and_predict` on a single evaluation set followed
    by :func:`score`. ``bootstrap`` (default 0 = off) is the number of bootstrap
    resamples for test-set confidence intervals.
    """
    records = fit_and_predict(model, train, {"test": test}, seed=seed)
    return score(records["test"], dataset_name=dataset_name, threshold=threshold,
                 bootstrap=bootstrap, cost_usd=cost_usd, extra_meta=extra_meta)
```

```python
"""Corpus registry: every dataset the study can train on or evaluate against.

One :class:`CorpusSpec` per corpus records how to load it, its native label set,
and how those labels project into the shared schemas. The pipeline and the
command line look corpora up here by name, so adding a corpus is one loader plus
one :func:`register` call, with no change to the runner.

Loaders import the HuggingFace ``datasets`` package lazily, so importing this
module needs neither that package nor the network.
"""

from __future__ import annotations

from collections.abc import Callable, Mapping
from dataclasses import dataclass
from typing import Optional

from . import labels as L
from .errors import ConfigError
from .loaders import EmotionDataset, load_goemotions, load_semeval2018
from .preprocessing import Projector

SCHEMAS: tuple = ("native", "ekman6", "sentiment3")


@dataclass(frozen=True)
class CorpusSpec:
    """How to load one corpus and project its labels."""

    name: str                 # registry key, e.g. "go_emotions"
    prefix: str               # results-file stem prefix, e.g. "go"
    loader: Callable[[], EmotionDataset]
    native_labels: tuple
    ekman_map: Optional[Mapping] = None      # native label -> Ekman label (None drops it)
    sentiment_map: Optional[Mapping] = None  # native label -> sentiment label
    multi_label: bool = True


CORPORA: dict = {}


def register(spec: CorpusSpec) -> CorpusSpec:
    """Add ``spec`` to the registry. A name can be registered only once."""
    if spec.name in CORPORA:
        raise ValueError(f"corpus {spec.name!r} is already registered")
    CORPORA[spec.name] = spec
    return spec


def get_corpus(name: str) -> CorpusSpec:
    """Look up a corpus by name."""
    try:
        return CORPORA[name]
    except KeyError:
        known = ", ".join(sorted(CORPORA))
        raise ConfigError(f"unknown corpus {name!r}; known corpora: {known}") from None


def corpus_names() -> list:
    """Names of every registered corpus, sorted."""
    return sorted(CORPORA)


def _projector(mapping: Mapping) -> Projector:
    def project(native_labels):
        return L.map_labels(native_labels, mapping)

    return project


def resolve_schema(corpus: str, schema: str):
    """Return ``(label_names, projector)`` for ``corpus`` under ``schema``.

    ``projector`` maps a list of native labels into the target schema; it is
    ``None`` for the ``native`` schema, meaning "use the corpus's labels as-is".
    """
    spec = get_corpus(corpus)
    if schema == "native":
        return list(spec.native_labels), None
    if schema == "ekman6":
        if spec.ekman_map is None:
            raise ConfigError(f"corpus {corpus!r} has no Ekman map")
        return [*L.EKMAN, L.NEUTRAL], _projector(spec.ekman_map)
    if schema == "sentiment3":
        if spec.sentiment_map is None:
            raise ConfigError(f"the sentiment3 schema is not defined for corpus {corpus!r}")
        return [*L.SENTIMENT, L.NEUTRAL], _projector(spec.sentiment_map)
    raise ConfigError(f"unknown schema {schema!r}; choose from {', '.join(SCHEMAS)}")


register(CorpusSpec(
    name="go_emotions", prefix="go", loader=load_goemotions,
    native_labels=tuple(L.GOEMOTIONS),
    ekman_map=L.GOEMOTIONS_TO_EKMAN, sentiment_map=L.GOEMOTIONS_TO_SENTIMENT,
))
register(CorpusSpec(
    name="sem_eval_2018_task_1", prefix="sem", loader=load_semeval2018,
    native_labels=tuple(L.SEMEVAL2018),
    ekman_map=L.SEMEVAL2018_TO_EKMAN,
))
```

```python
"""Model registry: every model configuration the study can run, by short name.

One :class:`ModelSpec` per configuration records how to build it, which *route*
to a lightweight model it belongs to, and whether its training is stochastic.
The pipeline and the command line build models through :func:`build_model`, so
adding a model is one :func:`register` call, with no change to the runner.

Routes (thesis spec, section 2.2): ``S`` small by design, ``F`` frozen encoder
plus linear head, ``C`` compressed transformer, ``ceiling`` full-size reference,
``hosted`` no-training hosted model. ``unassigned`` marks a HuggingFace
checkpoint requested ad hoc as ``hf_<checkpoint>`` and not yet classified.

Builders import their model classes lazily, so importing this module needs
neither torch nor transformers.
"""

from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass

from ..errors import ConfigError

ROUTES: tuple = ("S", "F", "C", "ceiling", "hosted", "unassigned")
HF_PREFIX = "hf_"


@dataclass(frozen=True)
class ModelSpec:
    """How to build one model configuration, and what kind of model it is."""

    name: str                    # registry key and command-line short name
    builder: Callable            # builder(**params) -> EmotionModel
    route: str                   # one of ROUTES
    stochastic: bool             # does training vary with the seed?
    takes_features: bool = False  # accepts features="tfidf" | "bow"
    takes_epochs: bool = False    # accepts epochs=<int>

    def __post_init__(self) -> None:
        if self.route not in ROUTES:
            raise ValueError(f"unknown route {self.route!r}; choose from {', '.join(ROUTES)}")


def _classical(class_name: str) -> Callable:
    def build(**params):
        from . import classical

        return getattr(classical, class_name)(**params)

    return build


def _bilstm(**params):
    from .deep import BiLSTMClassifier

    return BiLSTMClassifier(**params)


def _transformer(checkpoint: str) -> Callable:
    def build(**params):
        from .transformer import TransformerClassifier

        return TransformerClassifier(model_name=checkpoint, **params)

    return build


MODELS: dict = {}
ALIASES: dict = {
    "hf_distilbert": "hf_distilbert-base-uncased",
    "transformer": "hf_distilbert-base-uncased",
}


def register(spec: ModelSpec) -> ModelSpec:
    """Add ``spec`` to the registry. A name can be registered only once."""
    if spec.name in MODELS:
        raise ValueError(f"model {spec.name!r} is already registered")
    MODELS[spec.name] = spec
    return spec


# The stochastic flags are read off results/paper_final/data/go_native_seeds_summary.csv:
# Naive Bayes, LogReg and LinearSVM show zero spread over five seeds; Random Forest,
# the BiLSTM and the transformers do not.
register(ModelSpec("naive_bayes", _classical("NaiveBayes"), route="S",
                   stochastic=False, takes_features=True))
register(ModelSpec("logreg", _classical("LogisticReg"), route="S",
                   stochastic=False, takes_features=True))
register(ModelSpec("linsvm", _classical("LinearSVM"), route="S",
                   stochastic=False, takes_features=True))
register(ModelSpec("random_forest", _classical("RandomForest"), route="S",
                   stochastic=True, takes_features=True))
register(ModelSpec("bilstm", _bilstm, route="S", stochastic=True, takes_epochs=True))
register(ModelSpec("hf_distilbert-base-uncased", _transformer("distilbert-base-uncased"),
                   route="C", stochastic=True, takes_epochs=True))
register(ModelSpec("hf_bert-base-uncased", _transformer("bert-base-uncased"),
                   route="ceiling", stochastic=True, takes_epochs=True))


def model_names() -> list:
    """Names of every registered model, sorted."""
    return sorted(MODELS)


def get_model_spec(name: str) -> ModelSpec:
    """Look up a model by short name, alias, or ``hf_<checkpoint>``."""
    name = ALIASES.get(name, name)
    if name in MODELS:
        return MODELS[name]
    if name.startswith(HF_PREFIX) and len(name) > len(HF_PREFIX):
        return ModelSpec(name, _transformer(name[len(HF_PREFIX):]), route="unassigned",
                         stochastic=True, takes_epochs=True)
    known = ", ".join(sorted(MODELS))
    raise ConfigError(f"unknown model {name!r}; known models: {known}, or hf_<checkpoint>")


def build_model(name: str, *, features: str = "tfidf", epochs=None, **params):
    """Instantiate a model by short name.

    ``features`` goes to models with a text-feature backend (the classical tier).
    ``epochs``, if set, overrides the training-epoch default of the models that
    train by epochs and is ignored by the rest. ``params`` are passed straight to
    the model's constructor.
    """
    spec = get_model_spec(name)
    kwargs = dict(params)
    if spec.takes_features:
        kwargs["features"] = features
    if spec.takes_epochs and epochs:
        kwargs["epochs"] = epochs
    return spec.builder(**kwargs)
```

```python
"""The experiment pipeline: train each model once and fan it out to many evaluation sets.

:func:`run_pipeline` takes a :class:`RunConfig`, loads the training corpus and
every corpus an evaluation set names, and for each (model, seed) trains once and
predicts on all evaluation sets. Each prediction is scored into a scorecard row
and, when a predictions directory is given, written to the prediction cache so
it can be re-scored later (tuned thresholds, calibration, significance tests)
without a retrain.

An evaluation set is named ``"<corpus>:<split>"``, for example
``go_emotions:test`` or ``sem_eval_2018_task_1:test``.
"""

from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional

import numpy as np

from .corpora import get_corpus, resolve_schema
from .errors import ConfigError
from .experiment import fit_and_predict, score
from .models.registry import build_model
from .predictions import prediction_path, save_prediction
from .preprocessing import PreparedSplit, prepare_dataset
from .scorecard import Scorecard


@dataclass
class RunConfig:
    """Everything that defines one pipeline run."""

    models: list
    dataset: str = "go_emotions"            # the training corpus
    schema: str = "native"
    features: str = "tfidf"
    train_split: str = "train"
    test_split: str = "test"
    # Evaluation sets to score, as "<corpus>:<split>". None means the training
    # corpus's test split.
    eval_sets: Optional[list] = None
    # Training-corpus split predicted and cached (not scored) for later tuning.
    # Only used when predictions_dir is set; None turns it off.
    val_split: Optional[str] = "validation"
    seeds: list = field(default_factory=lambda: [42])
    epochs: Optional[int] = None
    limit_train: Optional[int] = None       # first N training rows (smoke runs)
    limit_eval: Optional[int] = None        # first N rows of every evaluation set
    subsample_train: Optional[int] = None   # random N training rows, redrawn per seed
    bootstrap: int = 0
    predictions_dir: Optional[Path] = None  # where to cache predictions; None = don't
    run_name: str = "run"


@dataclass
class RunResult:
    """What a pipeline run produced."""

    card: Scorecard
    prediction_paths: list = field(default_factory=list)


def parse_eval_set(name: str) -> tuple:
    """Split ``"<corpus>:<split>"`` into ``(corpus, split)``."""
    corpus, sep, split = name.partition(":")
    if not sep or not corpus or not split:
        raise ConfigError(f"an evaluation set is named '<corpus>:<split>', got {name!r}")
    return corpus, split


def head(split: PreparedSplit, n: int) -> PreparedSplit:
    """The first ``n`` rows of ``split`` as a new object; the input is not modified."""
    return PreparedSplit(split.texts[:n], split.Y[:n], list(split.label_names))


def subsample(split: PreparedSplit, n: int, seed: int) -> PreparedSplit:
    """A random ``n``-row subset of ``split``, drawn with ``seed``.

    The draw is ``np.random.RandomState(seed).choice`` without replacement, rows
    kept in their original order: the same draw the matched-size control runs
    used. If ``n`` is not smaller than the split, the split is returned as-is.
    The input is never modified.
    """
    if n >= len(split):
        return split
    rng = np.random.RandomState(seed)
    idx = np.sort(rng.choice(len(split), size=n, replace=False))
    return PreparedSplit([split.texts[i] for i in idx], split.Y[idx], list(split.label_names))


def dataset_label(cfg: RunConfig, eval_set: str) -> str:
    """The scorecard ``dataset`` value for one evaluation set.

    The training corpus's test split is just the corpus name; another corpus is
    ``"<train>-><corpus>"``; any split other than the test split is appended as
    ``":<split>"``.
    """
    corpus, split = parse_eval_set(eval_set)
    label = cfg.dataset if corpus == cfg.dataset else f"{cfg.dataset}->{corpus}"
    return label if split == cfg.test_split else f"{label}:{split}"


def run_pipeline(cfg: RunConfig, *, log: Optional[Callable] = None) -> RunResult:
    """Run every model in ``cfg`` over every seed and evaluation set.

    ``log``, if given, is called with one progress line at a time (the command
    line passes ``print``). Configuration problems raise :class:`ConfigError`
    before any model is trained.
    """
    say = log if log is not None else (lambda message: None)
    if not cfg.models:
        raise ConfigError("RunConfig.models must name at least one model")
    if not cfg.seeds:
        raise ConfigError("RunConfig.seeds must hold at least one seed")

    label_names, _ = resolve_schema(cfg.dataset, cfg.schema)
    prepared_cache: dict = {}

    def prepared(corpus: str) -> dict:
        if corpus not in prepared_cache:
            names, projector = resolve_schema(corpus, cfg.schema)
            if names != label_names:
                raise ConfigError(
                    f"corpus {corpus!r} does not share a label space with "
                    f"{cfg.dataset!r} under the {cfg.schema!r} schema; use a shared "
                    f"schema such as ekman6"
                )
            dataset = get_corpus(corpus).loader()
            prepared_cache[corpus] = prepare_dataset(dataset, names, projector)
        return prepared_cache[corpus]

    def take(corpus: str, split: str) -> PreparedSplit:
        splits = prepared(corpus)
        if split not in splits:
            raise ConfigError(
                f"corpus {corpus!r} has no split {split!r}; it has: {', '.join(splits)}"
            )
        return splits[split]

    def capped(split: PreparedSplit) -> PreparedSplit:
        return head(split, cfg.limit_eval) if cfg.limit_eval else split

    train = take(cfg.dataset, cfg.train_split)
    if cfg.limit_train:
        train = head(train, cfg.limit_train)

    report_names = list(cfg.eval_sets) if cfg.eval_sets else [f"{cfg.dataset}:{cfg.test_split}"]
    if len(set(report_names)) != len(report_names):
        raise ConfigError(f"duplicate evaluation sets: {', '.join(report_names)}")
    eval_sets = {name: capped(take(*parse_eval_set(name))) for name in report_names}

    # The training corpus's validation split is predicted and cached but not scored:
    # it is there so thresholds and calibration can be tuned later without a retrain.
    val_name = None
    if cfg.predictions_dir and cfg.val_split:
        candidate = f"{cfg.dataset}:{cfg.val_split}"
        if candidate not in eval_sets and cfg.val_split in prepared(cfg.dataset):
            val_name = candidate
            eval_sets[val_name] = capped(take(cfg.dataset, cfg.val_split))

    sizes = "  ".join(f"{name}={len(split)}" for name, split in eval_sets.items())
    say(f"  train={len(train)}  n_labels={len(label_names)}  {sizes}")
    if cfg.subsample_train and cfg.subsample_train < len(train):
        say(f"  each seed trains on a random {cfg.subsample_train}-row subsample")

    result = RunResult(Scorecard())
    multiseed = len(cfg.seeds) > 1
    for model_name in cfg.models:
        say(f"\n=== {model_name} ===")
        for seed in cfg.seeds:
            fit_train = train
            if cfg.subsample_train:
                fit_train = subsample(train, cfg.subsample_train, seed)
            model = build_model(model_name, features=cfg.features, epochs=cfg.epochs)
            records = fit_and_predict(model, fit_train, eval_sets, seed=seed, meta={
                "run_name": cfg.run_name,
                "train_corpus": cfg.dataset,
                "train_split": cfg.train_split,
                "schema": cfg.schema,
                "features": cfg.features,
                "epochs": cfg.epochs,
                "subsample_train": cfg.subsample_train,
            })
            for eval_name, record in records.items():
                if cfg.predictions_dir:
                    path = prediction_path(cfg.predictions_dir, cfg.run_name,
                                           record.meta["model"], seed, eval_name)
                    result.prediction_paths.append(save_prediction(record, path))
                if eval_name == val_name:
                    continue
                row = score(record, dataset_name=dataset_label(cfg, eval_name),
                            bootstrap=cfg.bootstrap, extra_meta={
                                "schema": cfg.schema,
                                "features": cfg.features,
                                "epochs": cfg.epochs,
                                "subsample_train": cfg.subsample_train,
                                "test_dataset": parse_eval_set(eval_name)[0],
                                "eval_set": eval_name,
                            })
                result.card.add(row)
                tag = f"[seed {seed}] " if multiseed else ""
                if len(report_names) > 1:
                    tag += f"[{row.dataset}] "
                say(f"  {tag}macro_f1={row.macro_f1:.3f}  micro_f1={row.micro_f1:.3f}  "
                    f"ece={row.ece:.3f}  train={row.train_seconds:.1f}s  "
                    f"latency={row.predict_latency_ms:.2f}ms  thrpt={row.throughput:.0f}/s  "
                    f"size={row.model_size_mb:.1f}MB  device={row.device}")
    return result
```

```python
r"""Run one or more models against a corpus and print a scorecard.

Each model is trained once per seed and then predicts on every evaluation set, so
one run can score the in-domain test split and any number of other corpora
without retraining. The per-example probabilities are cached next to the results.

Examples
--------
    # Lightweight tier on GoEmotions, full native 27-class taxonomy (the default)
    python scripts/run_experiment.py \
        --models naive_bayes logreg linsvm random_forest \
        --dataset go_emotions --limit-train 4000 --limit-test 1500

    # Bag-of-Words features instead of TF-IDF
    python scripts/run_experiment.py --models logreg --features bow

    # Add the BiLSTM tier and the transformer reference
    python scripts/run_experiment.py --models logreg bilstm hf_distilbert

    # Granularity sweep: same models under a coarser schema
    python scripts/run_experiment.py --models logreg --schema ekman6

    # One fit, two evaluation sets: in-domain test plus cross-corpus transfer
    python scripts/run_experiment.py --models logreg --schema ekman6 \
        --eval-sets go_emotions:test sem_eval_2018_task_1:test

Label schema (`--schema`) controls the target label space:
    native      corpus-native labels (GoEmotions = 27 emotions + neutral) [default]
    ekman6      collapsed Ekman-6 (+ neutral)
    sentiment3  collapsed positive/negative/ambiguous (+ neutral); GoEmotions only

Evaluation sets (`--eval-sets`) are named `<corpus>:<split>`. The default is the
training corpus's test split. The training corpus's validation split is also
predicted and cached (never scored) so thresholds can be tuned later; turn that
off with `--val-split none`. Predictions go to `<out-dir>/predictions/<run-name>/`.
"""

from __future__ import annotations

import argparse
from pathlib import Path

from emotion_classification.corpora import SCHEMAS, corpus_names
from emotion_classification.errors import ConfigError
from emotion_classification.models.registry import model_names
from emotion_classification.pipeline import RunConfig, run_pipeline
from emotion_classification.results import save_results, save_summary, summarize


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description=__doc__,
                                     formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("--models", nargs="+", default=["logreg"],
                        help="model short names: " + ", ".join(model_names()) +
                             ", or hf_<checkpoint> (e.g. hf_distilbert)")
    parser.add_argument("--dataset", default="go_emotions", choices=corpus_names(),
                        help="training corpus (default: go_emotions)")
    parser.add_argument("--schema", default="native", choices=list(SCHEMAS),
                        help="target label space (default: native)")
    parser.add_argument("--features", default="tfidf", choices=["tfidf", "bow"],
                        help="feature backend for classical models (default: tfidf)")
    parser.add_argument("--train-split", default="train")
    parser.add_argument("--test-split", default="test")
    parser.add_argument("--eval-sets", nargs="+", default=None, metavar="CORPUS:SPLIT",
                        help="evaluation sets to score; each model is trained once and "
                             "predicts on all of them (default: <dataset>:<test-split>)")
    parser.add_argument("--test-dataset", default=None, choices=corpus_names(),
                        help="shorthand for --eval-sets <corpus>:<test-split> (cross-dataset "
                             "generalization). Requires a shared schema, e.g. ekman6.")
    parser.add_argument("--val-split", default="validation",
                        help="training-corpus split whose predictions are cached, unscored, "
                             "for later tuning (default: validation; 'none' turns it off)")
    parser.add_argument("--limit-train", type=int, default=None,
                        help="cap training rows (speeds up smoke runs)")
    parser.add_argument("--limit-test", "--limit-eval", dest="limit_eval", type=int,
                        default=None, help="cap the rows of every evaluation set")
    parser.add_argument("--seed", type=int, default=42)
    parser.add_argument("--seeds", nargs="+", type=int, default=None,
                        help="run each model across these seeds and report mean±std "
                             "(overrides --seed; e.g. --seeds 42 43 44 45 46)")
    parser.add_argument("--out-dir", default="results",
                        help="directory for persisted results (default: results/; thesis "
                             "batches pass results/thesis/<batch>)")
    parser.add_argument("--run-name", default=None,
                        help="basename for saved files (default: <dataset>_<schema>_<features>)")
    parser.add_argument("--no-save", action="store_true",
                        help="skip writing results and predictions to disk")
    parser.add_argument("--no-predictions", action="store_true",
                        help="write the scorecard files but not the prediction cache")
    parser.add_argument("--bootstrap", type=int, default=0,
                        help="bootstrap resamples for test-set F1 confidence intervals "
                             "(0 = off; e.g. 1000). Adds macro/micro + per-emotion CIs.")
    parser.add_argument("--epochs", type=int, default=None,
                        help="training epochs for deep/transformer tiers (overrides the "
                             "model default; e.g. 5). Ignored by classical models.")
    parser.add_argument("--subsample-train", type=int, default=None,
                        help="randomly sample this many training rows before fitting, for "
                             "matched-size control experiments. Each seed draws its own "
                             "subsample. Unlike --limit-train (first N), this is random.")
    return parser


def config_from_args(args: argparse.Namespace) -> RunConfig:
    """Translate parsed arguments into a :class:`RunConfig`."""
    if args.eval_sets and args.test_dataset:
        raise SystemExit("use either --eval-sets or --test-dataset, not both")
    eval_sets = args.eval_sets
    if args.test_dataset:
        eval_sets = [f"{args.test_dataset}:{args.test_split}"]
    save_predictions = not (args.no_save or args.no_predictions)
    return RunConfig(
        models=list(args.models),
        dataset=args.dataset,
        schema=args.schema,
        features=args.features,
        train_split=args.train_split,
        test_split=args.test_split,
        eval_sets=eval_sets,
        val_split=None if args.val_split.lower() == "none" else args.val_split,
        seeds=list(args.seeds) if args.seeds else [args.seed],
        epochs=args.epochs,
        limit_train=args.limit_train,
        limit_eval=args.limit_eval,
        subsample_train=args.subsample_train,
        bootstrap=args.bootstrap,
        predictions_dir=Path(args.out_dir) / "predictions" if save_predictions else None,
        run_name=args.run_name or f"{args.dataset}_{args.schema}_{args.features}",
    )


def main(argv=None) -> int:
    args = build_parser().parse_args(argv)
    cfg = config_from_args(args)

    print(f"Loading {cfg.dataset} (schema={cfg.schema}, features={cfg.features}) ...")
    try:
        result = run_pipeline(cfg, log=print)
    except ConfigError as err:
        raise SystemExit(str(err)) from None
    card = result.card
    multiseed = len(cfg.seeds) > 1

    print("\n## Raw scorecard\n")
    print(card.to_markdown())

    if multiseed:
        print(f"\n## Multi-seed summary (mean +/- std over {len(cfg.seeds)} seeds)\n")
        for s in summarize(card):
            print(f"  {s['model']} [{s['dataset']}]:  "
                  f"macro_f1={s['macro_f1_mean']:.3f}+/-{s['macro_f1_std']:.3f}  "
                  f"micro_f1={s['micro_f1_mean']:.3f}+/-{s['micro_f1_std']:.3f}  "
                  f"train={s['train_seconds_mean']:.1f}+/-{s['train_seconds_std']:.1f}s  "
                  f"size={s['model_size_mb_mean']:.1f}MB")
    elif len(card) > 1:
        print("\n## Normalized (1 = best per axis)\n")
        norm = card.normalized()
        axes = card.axes()
        header = ["model", "dataset", *axes]
        print("| " + " | ".join(header) + " |")
        print("| " + " | ".join("---" for _ in header) + " |")
        for r in norm:
            cells = [r["model"], r["dataset"], *[f"{r.get(a, float('nan')):.3f}" for a in axes]]
            print("| " + " | ".join(cells) + " |")

    if not args.no_save:
        paths = save_results(card, args.out_dir, cfg.run_name)
        if multiseed:
            paths["summary"] = save_summary(card, args.out_dir, cfg.run_name)
        print("\nSaved results:")
        for kind, path in paths.items():
            print(f"  {kind:<11} {path}")
        if result.prediction_paths:
            print(f"  {'predictions':<11} {len(result.prediction_paths)} files in "
                  f"{result.prediction_paths[0].parent}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

````markdown
Each model is trained once and evaluated on every evaluation set, so one run can
hold several rows per model. The `dataset` column tells them apart:

| Value | Meaning |
|---|---|
| `go_emotions` | trained and tested on GoEmotions (its test split) |
| `go_emotions->sem_eval_2018_task_1` | trained on GoEmotions, tested on SemEval-2018 |

Every `.npz` file holds `y_true`, `y_prob`, the label names, and a manifest

device). Scores are recomputed from these files exactly, so thresholds,
````

```python
"""Regression against the published table.

Needs the GoEmotions download and trains two classical models on the full corpus
(about a minute), so it is marked ``network`` and ``slow`` and does not run in CI:

    python -m pytest -m "network and slow" tests/test_regression_paper.py
"""

import csv

import pytest

from emotion_classification.config import PROJECT_ROOT
from emotion_classification.experiment import score
from emotion_classification.pipeline import RunConfig, run_pipeline
from emotion_classification.predictions import load_prediction

PAPER_TABLE = PROJECT_ROOT / "results" / "paper_final" / "data" / "go_native_final_scorecard.csv"

# The table was produced on the GPU server. A different scikit-learn build on
# another machine moves these deterministic models by a few 1e-4 (measured:
# at most 5e-4), far below any difference the study interprets.
TOLERANCE = 2e-3


@pytest.mark.network
@pytest.mark.slow
def test_classical_rows_match_the_published_table(tmp_path):
    pytest.importorskip("datasets")
    with PAPER_TABLE.open(newline="", encoding="utf-8") as fh:
        paper = {row["model"]: row for row in csv.DictReader(fh)}

    cfg = RunConfig(models=["naive_bayes", "logreg"], dataset="go_emotions",
                    schema="native", predictions_dir=tmp_path, run_name="regression")
    result = run_pipeline(cfg)

    assert [row.model for row in result.card.rows] == ["naive_bayes_tfidf", "logreg_tfidf"]
    for row in result.card.rows:
        published = paper[row.model]
        assert row.dataset == "go_emotions"
        assert row.meta["n_train"] == int(published["n_train"])
        assert row.meta["n_test"] == int(published["n_test"])
        for axis in ("macro_f1", "micro_f1", "subset_accuracy"):
            assert getattr(row, axis) == pytest.approx(float(published[axis]), abs=TOLERANCE)

    # Validation and test were cached for both models ...
    assert sorted(path.name for path in result.prediction_paths) == [
        "logreg_tfidf__seed42__go_emotions-test.npz",
        "logreg_tfidf__seed42__go_emotions-validation.npz",
        "naive_bayes_tfidf__seed42__go_emotions-test.npz",
        "naive_bayes_tfidf__seed42__go_emotions-validation.npz",
    ]
    # ... and re-scoring the cache gives the scorecard row back exactly.
    by_model = {row.model: row for row in result.card.rows}
    for path in result.prediction_paths:
        record = load_prediction(path)
        if record.meta["eval_set"] == "go_emotions:validation":
            assert len(record) == 5426
            continue
        row = by_model[record.meta["model"]]
        again = score(record, dataset_name="go_emotions", extra_meta={
            key: row.meta[key] for key in
            ("schema", "features", "epochs", "subsample_train", "test_dataset", "eval_set")
        })
        assert again.as_dict() == row.as_dict()
```
